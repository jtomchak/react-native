cmake_minimum_required(VERSION 3.22)
project(nativehost_android LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(NH_TEST_HOOKS "Enable explicit test build hooks" OFF)
option(NH_DEV_UPDATES "Enable isolated development updates" OFF)
option(NH_PRODUCTION_ARTIFACT "Build a production app artifact" OFF)
if(NH_PRODUCTION_ARTIFACT AND NH_DEV_UPDATES)
  message(FATAL_ERROR "Development updates are forbidden in production artifacts")
endif()
get_filename_component(NH_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../.." ABSOLUTE)
get_filename_component(REPOSITORY_ROOT "${NH_ROOT}/../.." ABSOLUTE)
find_package(hermes-engine REQUIRED CONFIG)
set(NH_JSI_INCLUDE "${NH_ROOT}/platforms/android/.cache/hermes-jsi" CACHE PATH "JSI headers from Hermes 250829098.0.16 distribution")
if(NOT EXISTS "${NH_JSI_INCLUDE}/jsi/hermes-interfaces.h")
  message(FATAL_ERROR "Run platforms/android/prepare-hermes-headers.py to obtain matched JSI headers")
endif()
foreach(JSI_HEADER jsi.h jsi-inl.h)
  file(SHA256 "${NH_JSI_INCLUDE}/jsi/${JSI_HEADER}" MATCHED_JSI_HASH)
  file(SHA256 "${REPOSITORY_ROOT}/packages/react-native/ReactCommon/jsi/jsi/${JSI_HEADER}" REPOSITORY_JSI_HASH)
  if(NOT MATCHED_JSI_HASH STREQUAL REPOSITORY_JSI_HASH)
    message(FATAL_ERROR "JSI header ${JSI_HEADER} differs from the matched Hermes distribution; resolve ABI version before building")
  endif()
endforeach()
add_subdirectory("${REPOSITORY_ROOT}/packages/react-native/ReactCommon/yoga/yoga" "${CMAKE_CURRENT_BINARY_DIR}/yoga")

find_program(CARGO cargo HINTS "$ENV{HOME}/.cargo/bin" REQUIRED)
set(RUST_BUILD "${CMAKE_CURRENT_BINARY_DIR}/rust")
set(RUST_LIBRARY "${RUST_BUILD}/aarch64-linux-android/release/libnative_host_core.a")
set(RUST_FEATURES "")
if(NH_TEST_HOOKS)
  set(RUST_FEATURES --features test-hooks)
endif()
file(GLOB_RECURSE RUST_SOURCES CONFIGURE_DEPENDS "${NH_ROOT}/core/src/*.rs")
add_custom_command(OUTPUT "${RUST_LIBRARY}"
  COMMAND ${CMAKE_COMMAND} -E env
    "CARGO_TARGET_AARCH64_LINUX_ANDROID_LINKER=${CMAKE_ANDROID_NDK}/toolchains/llvm/prebuilt/darwin-x86_64/bin/aarch64-linux-android26-clang"
    ${CARGO} build --manifest-path "${NH_ROOT}/core/Cargo.toml" --target aarch64-linux-android --release --target-dir "${RUST_BUILD}" ${RUST_FEATURES}
  DEPENDS ${RUST_SOURCES} "${NH_ROOT}/core/Cargo.toml"
  VERBATIM)
add_custom_target(native_host_rust DEPENDS "${RUST_LIBRARY}")
add_library(jsi SHARED "${REPOSITORY_ROOT}/packages/react-native/ReactCommon/jsi/jsi/jsi.cpp")
target_include_directories(jsi PRIVATE "${NH_JSI_INCLUDE}")
add_library(nativehost_android SHARED bridge.cpp "${NH_ROOT}/runtime/Host.cpp")
add_dependencies(nativehost_android native_host_rust)
target_include_directories(nativehost_android PRIVATE "${NH_ROOT}/runtime" "${NH_ROOT}/core/include" "${NH_JSI_INCLUDE}")
target_link_libraries(nativehost_android PRIVATE hermes-engine::hermesvm jsi yogacore "${RUST_LIBRARY}" android log)
if(NH_TEST_HOOKS)
  target_compile_definitions(nativehost_android PRIVATE NH_TEST_HOOKS=1)
endif()
if(NH_DEV_UPDATES)
  target_compile_definitions(nativehost_android PRIVATE NH_DEV_UPDATES=1)
endif()
if(NH_PRODUCTION_ARTIFACT)
  target_compile_definitions(nativehost_android PRIVATE NH_PRODUCTION_ARTIFACT=1)
endif()
