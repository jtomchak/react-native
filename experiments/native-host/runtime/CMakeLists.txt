cmake_minimum_required(VERSION 3.22)
project(native_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(NH_DEV_UPDATES "Compile development-only immutable update ingress" OFF)
option(NH_PRODUCTION_ARTIFACT "Forbid development ingress in production artifacts" OFF)
if(NH_PRODUCTION_ARTIFACT AND NH_DEV_UPDATES)
  message(FATAL_ERROR "NH_DEV_UPDATES is forbidden in production artifacts")
endif()
option(NH_TEST_HOOKS "Compile explicit test-only evaluation hooks" OFF)
option(NH_SERVICE_TEST_HOOKS "Compile test-only Rust error injection (core test-hooks feature required)" OFF)
get_filename_component(PROTOTYPE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
get_filename_component(REPOSITORY_ROOT "${PROTOTYPE_ROOT}/../.." ABSOLUTE)
set(HERMES_ROOT "" CACHE PATH "Matched Hermes destroot with include and frameworks")
set(HERMES_FRAMEWORK "" CACHE PATH "Matched Hermes framework directory")
set(NH_CORE_LIBRARY "${PROTOTYPE_ROOT}/core/target/release/libnative_host_core.a" CACHE FILEPATH "Rust core archive")
if(NOT EXISTS "${HERMES_ROOT}/include/hermes/hermes.h")
  message(FATAL_ERROR "Set HERMES_ROOT to the matched Hermes destroot")
endif()
add_subdirectory("${REPOSITORY_ROOT}/packages/react-native/ReactCommon/yoga/yoga" "${CMAKE_CURRENT_BINARY_DIR}/yoga")
add_library(native_host_runtime STATIC Host.cpp)
target_include_directories(native_host_runtime PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${PROTOTYPE_ROOT}/core/include" "${HERMES_ROOT}/include")
target_link_libraries(native_host_runtime PUBLIC yogacore "${NH_CORE_LIBRARY}")
if(NH_PRODUCTION_ARTIFACT)
  target_compile_definitions(native_host_runtime PUBLIC NH_PRODUCTION_ARTIFACT=1)
endif()
if(NH_DEV_UPDATES)
  target_compile_definitions(native_host_runtime PUBLIC NH_DEV_UPDATES=1)
endif()
if(NH_TEST_HOOKS)
  target_compile_definitions(native_host_runtime PRIVATE NH_TEST_HOOKS=1)
endif()
if(NH_SERVICE_TEST_HOOKS)
  if(NOT NH_TEST_HOOKS)
    message(FATAL_ERROR "NH_SERVICE_TEST_HOOKS requires NH_TEST_HOOKS")
  endif()
  target_compile_definitions(native_host_runtime PRIVATE NH_SERVICE_TEST_HOOKS=1)
endif()
if(APPLE)
  if(NOT EXISTS "${HERMES_FRAMEWORK}/hermesvm")
    message(FATAL_ERROR "Set HERMES_FRAMEWORK to the matched platform hermesvm.framework")
  endif()
  target_link_libraries(native_host_runtime PUBLIC "${HERMES_FRAMEWORK}/hermesvm" "-framework Foundation")
  get_filename_component(HERMES_FRAMEWORK_PARENT "${HERMES_FRAMEWORK}" DIRECTORY)
  target_link_options(native_host_runtime PUBLIC "-Wl,-rpath,${HERMES_FRAMEWORK_PARENT}")
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS" AND NOT ANDROID)
  add_executable(native-host main.cpp)
  target_link_libraries(native-host PRIVATE native_host_runtime)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS" AND NOT ANDROID)
  enable_testing()
  add_executable(host-contract-tests host-contract-tests.cpp)
  target_link_libraries(host-contract-tests PRIVATE native_host_runtime)
  # Contract assertions must remain active even when dependencies build Release.
  target_compile_options(host-contract-tests PRIVATE -UNDEBUG)
  if(NH_TEST_HOOKS)
    target_compile_definitions(host-contract-tests PRIVATE NH_TEST_HOOKS=1)
  endif()
  add_test(NAME host-contracts COMMAND host-contract-tests)
endif()

if(NH_DEV_UPDATES AND NOT CMAKE_SYSTEM_NAME STREQUAL "iOS" AND NOT ANDROID)
  add_executable(dev-contract-tests dev/contracts.cpp)
  target_link_libraries(dev-contract-tests PRIVATE native_host_runtime)
  target_compile_options(dev-contract-tests PRIVATE -UNDEBUG)
  add_test(NAME dev-contracts COMMAND dev-contract-tests)
  add_executable(native-host-dev-driver dev/driver.cpp)
  target_link_libraries(native-host-dev-driver PRIVATE native_host_runtime)
  set_tests_properties(dev-contracts PROPERTIES TIMEOUT 20)
  find_program(NH_NODE_EXECUTABLE node REQUIRED)
  add_test(NAME dev-integrity-node COMMAND "${NH_NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/dev/integrity.test.mjs" "$<TARGET_FILE:native-host-dev-driver>")
  set_tests_properties(dev-integrity-node PROPERTIES TIMEOUT 20)
endif()
